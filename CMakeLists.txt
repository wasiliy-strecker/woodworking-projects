cmake_minimum_required(VERSION 3.25)
project(HolzwerkProjekt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(HOLZWERKPROJEKT_SANITIZERS "Enable address and undefined behavior checks" OFF)
include(CTest)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Sql LinguistTools)
if(BUILD_TESTING)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
endif()

add_library(project_options INTERFACE)
if(MSVC)
  target_compile_options(project_options INTERFACE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
  if(HOLZWERKPROJEKT_SANITIZERS)
    target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(project_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(project_i18n src/i18n/languages.cpp src/i18n/privacy_notice.cpp)
target_include_directories(project_i18n PUBLIC src)
target_link_libraries(project_i18n PUBLIC Qt6::Core PRIVATE project_options)
set(app_languages de-DE en-US es-ES fr-FR it-IT pt-BR)
foreach(language IN LISTS app_languages)
  list(APPEND app_ts_files "${CMAKE_CURRENT_SOURCE_DIR}/translations/app_${language}.ts")
endforeach()
qt_add_lrelease(TS_FILES ${app_ts_files} QM_FILES_OUTPUT_VARIABLE app_qm_files)
qt_add_resources(project_i18n app_translations PREFIX "/i18n"
  BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES ${app_qm_files})
foreach(language de es fr it pt_BR)
  list(APPEND qt_qm_files "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_TRANSLATIONS}/qtbase_${language}.qm")
endforeach()
qt_add_resources(project_i18n qt_translations PREFIX "/i18n"
  BASE "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_TRANSLATIONS}" FILES ${qt_qm_files})

add_library(project_domain src/domain/project.cpp src/application/project_service.cpp)
target_include_directories(project_domain PUBLIC src)
target_link_libraries(project_domain PUBLIC project_i18n PRIVATE project_options)
add_library(project_photo src/application/photo_importer.cpp src/application/colors.cpp)
target_include_directories(project_photo PUBLIC src)
target_link_libraries(project_photo PUBLIC Qt6::Gui PRIVATE project_options)
add_library(project_storage src/application/collection_codec.cpp src/application/collection_transfer.cpp src/persistence/json_collection_file.cpp src/persistence/sqlite_project_repository.cpp src/persistence/data_directory.cpp src/persistence/language_preferences.cpp)
target_link_libraries(project_storage PUBLIC project_domain project_photo Qt6::Sql PRIVATE project_options)
add_library(project_ui src/ui/color_field.cpp src/ui/information_dialog.cpp src/ui/sort_header.cpp src/ui/export_file_dialog.cpp src/ui/checklist_editor.cpp src/ui/storage_controller.cpp src/ui/project_editor.cpp src/ui/main_window.cpp src/ui/ui_support.cpp src/ui/photo_view.cpp src/ui/language_dialog.cpp)
target_link_libraries(project_ui PUBLIC project_storage Qt6::Widgets PRIVATE project_options)
qt_add_executable(holzwerkprojekt WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(holzwerkprojekt PRIVATE project_storage project_ui project_options)
target_compile_definitions(holzwerkprojekt PRIVATE APP_VERSION="${PROJECT_VERSION}")
file(GLOB_RECURSE app_license_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/licenses/*")
qt_add_resources(project_ui app_notices PREFIX "/notices" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES THIRD_PARTY_NOTICES.md ${app_license_files})
qt_add_resources(holzwerkprojekt branding PREFIX "/" FILES assets/branding/icon-256.png)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
  target_sources(holzwerkprojekt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
set_target_properties(holzwerkprojekt PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER de.appfabrik.holzwerkprojekt
  MACOSX_BUNDLE_BUNDLE_NAME "Woodworking Projects"
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in")

if(BUILD_TESTING)
  foreach(test_name color transfer transfer_ui information collection_export_dialogs collection_features domain repository dialogs photo status photo_dialogs checklist i18n)
    add_executable(${test_name}_tests tests/${test_name}_test.cpp)
    target_link_libraries(${test_name}_tests PRIVATE project_storage project_ui Qt6::Test project_options)
    add_test(NAME ${test_name} COMMAND ${test_name}_tests)
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME catalogs COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_translations.py")
endif()

install(TARGETS holzwerkprojekt BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION licenses)
install(DIRECTORY licenses/ DESTINATION licenses)
if(WIN32 OR APPLE)
  qt_generate_deploy_app_script(TARGET holzwerkprojekt OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
